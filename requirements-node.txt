# Python dependencies for each Apple Silicon cluster node.
#
# `scripts/bootstrap-node.sh` installs these inside ~/mlx-cluster/.venv
# on each node. Pinning the versions keeps `runner.py` stable against
# upstream changes — bump intentionally, not by accident.
#
# Tested against the orchestrator at the version of api.py shipped in
# this repo. If you bump these, re-run bench + at least one multi-node
# load on a 100B+ MoE before merging.

mlx==0.32.0
mlx-lm==0.31.3

# Tokenizer support for the Kimi family (K2, Kimi-Linear, K3): their
# tokenizer is tiktoken-based custom code. Without these the model load
# dies at the tokenizer with a misleading "ranks died" — the failure is
# not persistent across nodes, so it belongs in this pinned set.
tiktoken>=0.13
blobfile>=3.2

# Hugging Face support — used by the runner when loading directly from
# a repo id (the dashboard's "Download from Hugging Face" path).
huggingface_hub>=0.28
